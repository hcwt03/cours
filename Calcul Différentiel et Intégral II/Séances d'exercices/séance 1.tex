 	\item 
	\begin{proof} Soient $((x_1,y_1),(x_2,y_2)) \in \R^2 \times \R^2$ et $\alpha \in \R$\\
		\underline{Inégalité triangulaire:}\\
			On a 
			\begin{align*}
			||(x_1+x_2,y_1+y_2)||_1 &= |x_1+x_2|+|y_1+y_2|\\
						&\leq (|x_1|+|y_1|)+(|x_2|+|y_2|)\ \text{par inégalité triangulaire de $|\cdot|$ sur $\R$}\\
						&= ||(x_1,y_1)||_1+||(x_2,y_2)||_1\\
			\end{align*}
			Comme $(x_1,y_1)$ et $(x_2,y_2)$ sont quelconques dans $\R^2$, l'inégalité est verifiée sur $\R^2 \times \R^2$.\\ 
		\underline{Linéarité pour la multiplication:}\\
			On a
			\begin{align*}
			||\alpha(x_1,y_1)||_1 &= ||(\alpha x_1,\alpha y_1)||_1\\
					      &= |\alpha x_1|+|\alpha y_1|\\
					      &= |\alpha|(|x_1|+|y_1|)\\
					      &= |\alpha| ||(x_1,y_2)||_1\\
			\end{align*}
			Comme $(x_1,y_1) \in \R^2$ et $\alpha \in \R$ sont quelconques, la linéarité est vérifiée sur $\R^2 \times \R$.
		\underline{$||(x,y)||_1=0 \iff (x,y)=(0,0)$:}\\
			\underline{$\implies$:}\\
				Par l'absurde, supposons $(x,y) \neq (0,0)$. Ceci implique $x \neq 0$ et $y \neq 0$ car, sinon, $||(x,y)||_1=|x|+|y|=0$ donnerait $x=y=0$.
				On a alors $|x|+|y|=0 \iff |x|=-|y|$. Or $\text{Im}(|\cdot|) = \R^+$, ce qui contredit le résultat précedent.
				Il en vient alors que $(x,y) = (0,0)$.\\
			\underline{$\Longleftarrow$:}
				On a $||(0,0)||_1=|0|+|0|=0$.
	\end{proof}

	\item
	\underline{$||\cdot||_2$:}
		\begin{enumerate}
		\item \begin{proof} Soient $((x_1,y_1),(x_2,y_2)) \in \R^2 \times \R^2$ et $\alpha \in \R$\\
			\underline{Inégalité triangulaire:}\\
			On peut d'abord remarquer que $(x_1y_2-y_1x_2)^2\geq 0\iff x_1^2y_2^2+y_1^2x_2^2\geq  2x_1x_2y_1y_2$.\\
			Il en suit alors, par croissance de $\sqrt{\cdot}$ sur $\R^+$, 
			\begin{align*}
				(||(x_1,y_1)||_2+||(x_2,y_2)||_2)^2 &= ||(x_1,y_1)||_2^2+||(x_2,y_2)||_2^2+2||(x_1,y_1)||_2||(x_2,y_2)||_2\\
								    &= x_1^2+x_2^2+y_1^2+y_2^2+2\sqrt{x_1^2x_2^2+x_1^2y_2^2+y_1^2x_2^2+y_1^2y_2^2}\\
								    &\geq x_1^2+x_2^2+y_1^2+y_2^2+2\sqrt{x_1^2x_2^2+y_1^2y_2^2+2x_1x_2y_1y_2}\\
								    &= x_1^2+x_2^2+y_1^2+y_2^2+2\sqrt{(x_1x_2+y_1y_2)^2}\\
								    &\geq x_1^2+x_2^2+y_1^2+y_2^2+2x_1x_2+2y_1y_2\\
								    &= (x_1+x_2)^2+(y_1+y_2)^2\\
								    &= ||(x_1+x_2,y_1+y_2)||_2^2
			\end{align*}
			À nouveau par croissance de $\sqrt{\cdot}$ sur $\R^+$, on déduit $||(x_1,y_1)||_2+||(x_2,y_2)||_2 \geq ||(x_1+x_2,y_1+y_2)||_2$.
			Comme $(x_1,y_1)$ et $(x_2,y_2)$ sont quelconques dans $\R^2$, l'inégalité est verifiée sur $\R^2 \times \R^2$.\\
			\underline{Linéarité pour la multiplication:}\\
			On a 
			\begin{align*}
				||\alpha(x_1,y_1)||_2 &= ||(\alpha x_1,\alpha y_1)||_2\\
						      &= \sqrt{(\alpha x_1)^2+(\alpha y_1)^2}\\
						      &= \sqrt{\alpha^2(x_1^2+y_1^2)}\\
						      &= |\alpha|\sqrt{x_1^2+y_1^2}\\
						      &= |\alpha|||(x_1,y_1)||_2
			\end{align*}
			Comme $(x_1,y_1) \in \R^2$ et $\alpha \in \R$ sont quelconques, la linéarité est vérifiée sur $\R^2 \times \R$.\\
			\underline{$||(x,y)||_2=0 \iff (x,y)=(0,0)$:}\\
			\underline{$\implies$:}\\
				Par l'absurde, supposons $(x,y) \neq (0,0)$. Ceci implique $x \neq 0$ et $y \neq 0$ car, sinon, $||(x,y)||_2=\sqrt{x^2+y^2}=0$, qui implique $x^2+y^2=0$ donnerait $x=y=0$.
				On a alors $\sqrt{x^2+y^2}=0 \iff x^2+y^2=0 \iff x=y=0$. Ce qui contredit le résultat précédent.
				Il en vient alors que $(x,y) = (0,0)$.\\
			\underline{$\Longleftarrow$:}
				On a $||(0,0)||_2=\sqrt{0^2+0^2}=0$.
		\end{proof}

	\underline{$||\cdot||_{\infty}$}
		\begin{enumerate}
		\item \begin{proof} Soient $((x_1,x_2),(x^*_1,x^*_2)) \in \R^2 \times \R^2$ et $\alpha \in \R$\\
                        \underline{Inégalité triangulaire:}\\
			Soit $i \in \{1;2\}$ tel que $||(x_1+x^*_1,x_2+x^*_2)||_{\infty}=|x_i+x^*_i|$. On a, par inégalité triangulaire de $|\cdot|$, 
			\begin{align*}
				||(x_1+x^*_1,x_2+x^*_2)||_{\infty} &= |x_i+x^*_i|\\
								   &\leq |x_i|+|x^*_i|\\
								   &\leq \max (\{ |x_1|;|x_2|\} )+\max (\{|x^*_1|;|x^*_2|\} )\\
								   &= |(x_1,x_2)|_{\infty}+|(x^*_1,x^*_2)|_{\infty}
			\end{align*}
			\underline{Linéarité pour la multiplication:}\\
			Soit $i \in \{1;2\}$ tel que $||(x_1,x_2)||_{\infty}=|x_i|$. On a
			\begin{align*}
				||\alpha(x_1,x_2)||_{\infty} &= ||(\alpha x_1,\alpha x_2)||_{\infty}\\
							     &= |\alpha x_i|\\
							     &= |\alpha||x_i|\\
							     &= |\alpha|||(x_1,x_2)||_{\infty}
			\end{align*}
			\underline{$||(x,y)||_{\infty}=0 \iff (x,y)=(0,0)$:}\\
			\underline{$\implies$:}\\
				On a, pour tout $i\in \{ 1;2\}$, $0\leq |x_i|\leq ||(x_1,x_2)||_{\infty}=0$. Ceci implique $x_1=x_2=0$.\\
			\underline{$\Longleftarrow$:}\\
			On a $||(0,0)||_{\infty}=\max (\{ 0;0\})=0$.
		\end{proof}

		\end{enumerate}
\end{enumerate}
